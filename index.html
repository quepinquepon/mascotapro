// Función para actualizar el selector de mascotas de forma segura function actualizarSelectorMascotas() { const select = document.getElementById('select-mascota-qr'); // Reemplaza con el ID real de tu select if (!select) return; let mascotas = JSON.parse(localStorage.getItem('mascotas_pro')) || []; select.innerHTML = ''; if (mascotas.length === 0) { // Estado vacío amigable (Ingeniería + UX) select.innerHTML += ''; return; } mascotas.forEach((m, index) => { let option = document.createElement('option'); option.value = index; option.textContent = m.nombre; select.appendChild(option); }); } // Ejecutar al cargar la página window.addEventListener('DOMContentLoaded', actualizarSelectorMascotas);
